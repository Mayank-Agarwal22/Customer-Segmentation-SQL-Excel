
1. SEGMENT & AGE ANALYSIS


select
      customer_segment,
      round(count(customer_name),1) as Total_customer,
      round(avg(customer_age),1) 
from customer_master 
group by customer_segment 
order by Total_customer desc;


2. REGIONAL DISTRIBUTION


select region,
       count(customer_name) as Total_customer 
from customer_master
group by region order by Total_customer desc;


3. TOP 10 TARGET STATES BY GENDER


select count(customer_name) as Total_customer,
       gender, customer_state 
from customer_master 
group by customer_state,gender
order by Total_customer desc limit 10;
